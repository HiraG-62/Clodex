withPending で、押したボタンを state が変わるまで pending にします。二度押しを止め、中身は CSS のスピナーに置き換えます。

ts · src/web/client/pending.ts
export const withPending = async <T>(
  button: HTMLButtonElement,
  run: () => Promise<T>,
): Promise<T | undefined> => {
  if (button.dataset.pending === PENDING) return undefined;
  button.dataset.pending = PENDING;
  button.classList.add("is-loading");
  button.setAttribute("aria-busy", "true");
  try {
    return await run();
  } finally {
    delete button.dataset.pending;
    button.classList.remove("is-loading");
    button.removeAttribute("aria-busy");
  }
};
